# 單一整合版工具：`doze_all-in-one.py`

您可以將以下內容直接儲存為 `~/doze/doze_all-in-one.py`。這個檔案**內建了完整的 ADB 自動偵測、連接埠掃描、裝置狀態擷取、BatteryStats 備份與進階分析功能**，讓您在手機上只需要維護和操作這單一檔案即可！

```python
#!/usr/bin/env python3
# -*- coding: utf-8 -*-

"""
Termux Android Doze & BatteryStats 全功能整合自動化工具 (All-in-One)

說明：
本工具將原本分散的 adb.sh、doze.sh 與 doze_summary-termux.py 完整整合為單一 Python 腳本。
無需 Root 權限，專為 Termux 設計，透過本機無線除錯自動連線、擷取快照並產生深度分析摘要。

用法：
  python3 doze_all-in-one.py [--port PORT] [--reset-batterystats] [--pkg PACKAGE_NAME]
"""

import argparse
import collections
import datetime as dt
import os
import re
import subprocess
import sys

# ============================================================
# 全域設定與常數
# ============================================================
WELL_KNOWN = {
    0: "root/kernel",
    1000: "system",
    1001: "radio/phone",
    1020: "mdnsd",
    1051: "netd",
    2000: "shell(adbd)",
}

BURST_MINUTES = 6
BURST_MAH = 9.0
HIGH_DRAIN_MAH_H = 25.0

DEFAULT_PKGS = ["ch.deletescape.lawnchair.plah"]


# ============================================================
# 第一部分：ADB 自動連線與環境管理 (原 adb.sh / doze.sh ADB 邏輯)
# ============================================================

def run_cmd(cmd, check=False):
    """執行外部 shell 指令並回傳標準輸出"""
    try:
        res = subprocess.run(cmd, shell=True, text=True, capture_output=True)
        if check and res.returncode != 0:
            return False, res.stderr.strip()
        return True, res.stdout.strip()
    except Exception as e:
        return False, str(e)


def check_port(port_str):
    if not port_str.isdigit():
        return False
    p = int(port_str)
    return 1 <= p <= 65535


def adb_is_connected(addr):
    ok, out = run_cmd("adb devices")
    if not ok:
        return False
    for line in out.splitlines():
        parts = line.split()
        if len(parts) >= 2 and parts[0] == addr and parts[1] == "device":
            return True
    return False


def find_connected_port():
    ok, out = run_cmd("adb devices")
    if not ok:
        return None
    for line in out.splitlines():
        match = re.match(r"^127\.0\.0\.1:(\d+)\s+device$", line)
        if match:
            return match.group(1)
    return None


def detect_adb_port():
    connected = find_connected_port()
    if connected:
        print(f"[ADB] 發現已有連線的端點 Port: {connected}")
        return connected

    print("[ADB] 正在使用 nmap 掃描本機無線除錯 Port (30000-65535)...")
    ok, _ = run_cmd("command -v nmap")
    if not ok:
        print("[ADB 警告] 找不到 nmap，無法自動掃描。")
        return None

    scan_cmd = "timeout 45s nmap -sT -Pn -n --open -T4 --host-timeout 45s -p 30000-65535 127.0.0.1 2>/dev/null"
    ok, out = run_cmd(scan_cmd)
    if not ok or not out:
        return None

    ports = re.findall(r"^(\d+)\/tcp\s+open", out, re.M)
    for p in ports:
        addr = f"127.0.0.1:{p}"
        run_cmd(f"adb connect {addr}")
        if adb_is_connected(addr):
            print(f"[ADB] 成功自動連線至端點: {addr}")
            return p

    return None


def ensure_adb_connected(manual_port=None):
    if manual_port:
        if not check_port(str(manual_port)):
            print(f"錯誤：指定的 Port {manual_port} 無效 (必須在 1-65535 之間)")
            sys.exit(1)
        port = manual_port
    else:
        port = detect_adb_port()
        if not port:
            print("\n無法自動偵測 ADB 連線埠。")
            try:
                port = input("請輸入手機「無線除錯」畫面顯示的 Port（直接按下 Enter 取消）： ").strip()
            except EOFError:
                port = ""
            if not port or not check_port(port):
                print("錯誤：未輸入有效的 Port，程式終止。")
                sys.exit(1)

    addr = f"127.0.0.1:{port}"
    print(f"[ADB] 正在建立連線至 {addr} ...")

    for attempt in range(1, 9):
        run_cmd(f"adb connect {addr}")
        if adb_is_connected(addr):
            print(f"[ADB] 連線成功：{addr}")
            return addr
        print(f"[ADB] 連線未就緒，重試中 ({attempt}/8)...")
        import time
        time.sleep(3)

    print(f"錯誤：無法連線至 {addr}，請確認手機無線除錯已開啟且 Port 正確。")
    sys.exit(1)


# ============================================================
# 第二部分：資料擷取與快照 (原 doze.sh 邏輯)
# ============================================================

def read_text(path):
    if not path:
        return ""
    try:
        with open(path, encoding="utf-8", errors="replace") as f:
            return f.read()
    except OSError:
        return ""


def parse_dur(value):
    units = {"d": 86400000, "h": 3600000, "m": 60000, "s": 1000, "ms": 1}
    total = 0
    for number, unit in re.findall(r"(\d+)\s*(ms|d|h|m|s)", value):
        total += int(number) * units[unit]
    return total


def fmt(milliseconds):
    milliseconds = max(0, int(milliseconds))
    return (
        f"{milliseconds // 3600000}h"
        f"{milliseconds % 3600000 // 60000:02d}m"
        f"{milliseconds % 60000 // 1000:02d}s"
    )


def grab(text, pattern, group=1, flags=re.M):
    match = re.search(pattern, text, flags)
    return match.group(group).strip() if match else None


def section(text, header, limit=80):
    pos = text.find(header)
    if pos < 0:
        return []
    return text[pos:].splitlines()[1:limit + 1]


def unique(items):
    return list(dict.fromkeys(items))


# ============================================================
# 第三部分：Python 摘要與分析引擎 (原 doze_summary-termux.py)
# ============================================================

def load_uidmap(path):
    result = collections.defaultdict(list)
    if not path:
        return result
    for line in read_text(path).splitlines():
        fields = line.strip().split(",")
        if len(fields) < 6:
            continue
        if fields[2] == "i" and fields[3] == "uid":
            try:
                uid = int(fields[4])
                result[uid].append(fields[5])
            except ValueError:
                continue
    return result


def uid_to_int(value):
    if value.isdigit():
        return int(value)
    match = re.match(r"u(\d+)a(\d+)$", value)
    if match:
        return int(match.group(1)) * 100000 + 10000 + int(match.group(2))
    return None


def label(uid_string, uidmap):
    uid = uid_to_int(uid_string)
    if uid is None:
        return uid_string
    if uid < 10000:
        if uid in WELL_KNOWN:
            return f"{uid_string}({WELL_KNOWN[uid]})"
        return uid_string
    names = uidmap.get(uid)
    if names:
        return f"{uid_string}({names[0]})"
    return uid_string


def per_uid_blocks(lines):
    current = None
    result = collections.defaultdict(list)
    for line in lines:
        match = re.match(r"^\s*(u\d+[a-z]\d+|\d+):\s*$", line)
        if match:
            current = match.group(1)
            continue
        if current and line.startswith("    "):
            result[current].append(line.strip())
        elif current and not line.startswith("    ") and line.strip():
            current = None
    return result


def split_dump(lines):
    for index, line in enumerate(lines):
        if line.startswith("Statistics since last charge"):
            return lines[:index], lines[index:]
    return lines, []


def parse_history(lines):
    events = []
    unparsed = 0
    for line in lines:
        match = re.match(r"^\s*(\+[\dhdms]+|0)\s+\((\d+)\)\s+(\d{1,3})\s?(.*)$", line)
        if not match:
            if line.strip():
                unparsed += 1
            continue
        timestamp, _, level, rest = match.groups()
        elapsed = 0 if timestamp == "0" else parse_dur(timestamp)
        events.append((elapsed, int(level), rest))
    events.sort(key=lambda item: item[0])
    return events, unparsed


def parse_idle(events):
    result = []
    for timestamp, _, rest in events:
        match = re.search(r"(?:^|\s)device_idle=([^\s]+)", rest)
        if match:
            result.append((timestamp, match.group(1)))
    return result


def parse_alarm_events(events):
    result = []
    for timestamp, _, rest in events:
        if "+alarm" not in rest:
            continue
        match = re.search(r'\+alarm=(\S+?):"([^"]*)"', rest)
        if match:
            result.append((timestamp, match.group(1), match.group(2)))
    return result


def analyze_and_print_report(full_text, checkin_text, deviceidle_text, mem_paths, battery_paths, pkgs_list):
    lines = full_text.splitlines()
    history_lines, stats_lines = split_dump(lines)
    stats_text = "\n".join(stats_lines)
    events, unparsed = parse_history(history_lines)
    uidmap = load_uidmap(checkin_text)

    print("\n# Android Doze / BatteryStats 摘要報告")
    print(f"- History 行數：{len(history_lines)}")
    print(f"- 成功解析事件：{len(events)}")
    print(f"- 未辨識 History 行：{unparsed}")

    start_dt = grab(stats_text, r"Start clock time:\s*(\S+)")
    print(f"- 統計起點：{start_dt or '未知'}")

    # 1. 電池統計與 CPU 狀態
    print("\n## 電池統計與 CPU 狀態")
    match = re.search(r"Time on battery:\s*(.*?)\s+\(([\d.]+)%\)\s+realtime,\s*(.*?)\s+\(([\d.]+)%\)\s+uptime", stats_text)
    if match:
        realtime, realtime_pct, uptime, uptime_pct = match.groups()
        print(f"- 電池使用時間：{realtime}")
        print(f"- CPU 清醒時間：{uptime}（{uptime_pct}%）")
        print(f"- CPU suspend 推算比例：{100 - float(uptime_pct):.1f}%")
    else:
        print("- ⚠ 找不到 Time on battery，無法計算 CPU 清醒比例。")

    fields = [
        ("Screen on", r"^\s+Screen on:\s*(.*)"),
        ("Discharge", r"^\s+Discharge:\s*(.*)"),
        ("Light doze discharge", r"Device light doze discharge:\s*(.*)"),
        ("Deep doze discharge", r"Device deep doze discharge:\s*(.*)"),
        ("Light idle", r"^\s+Device light idling:\s*(.*)"),
        ("Deep idle", r"^\s+Device full idling:\s*(.*)"),
        ("Partial wakelock", r"Total partial wakelock time:\s*(.*)"),
    ]
    for name, pattern in fields:
        val = grab(stats_text, pattern)
        if val:
            print(f"- {name}：{val}")

    # 2. Doze 時間軸
    print("\n## Doze 狀態與時間軸")
    idle = parse_idle(events)
    if idle:
        print("- device_idle 轉換：", ", ".join(f"{fmt(t)}={state}" for t, state in idle))
    else:
        print("- 沒有辨識到 device_idle 狀態事件。")

    # 3. 估算耗電
    print("\n## 估算耗電與指定套件分析")
    uid_rows = []
    for line in section(stats_text, "Estimated power use (mAh):", 80):
        m = re.match(r"^\s+[Uu][Ii][Dd] (\S+): ([\d.]+)", line)
        if m:
            uid_rows.append((float(m.group(2)), m.group(1)))
    uid_rows.sort(reverse=True)
    
    print("- UID 耗電前 8 名：")
    for val, uid in uid_rows[:8]:
        print(f"  - {label(uid, uidmap)}：{val:.2f} mAh")

    if pkgs_list:
        print("\n## 指定套件檢視")
        reverse = collections.defaultdict(list)
        for uid, names in uidmap.items():
            for name in names:
                reverse[name].append(uid)
        for pkg in pkgs_list:
            matches = reverse.get(pkg)
            if matches:
                uid = matches[0]
                print(f"- 套件 {pkg} (UID {uid}) 對應標籤：{label(str(uid), uidmap)}")

    print("\n摘要分析完成。")


# ============================================================
# 第四部分：主流程控制 (自動化測試流程)
# ============================================================

def main():
    parser = argparse.ArgumentParser(description="Android Doze & BatteryStats All-in-One 工具")
    parser.add_argument("port", nargs="?", help="手動指定的無線除錯 Port")
    parser.add_argument("--reset-batterystats", action="store_true", help="是否重置電池統計")
    parser.add_argument("--pkg", default="ch.deletescape.lawnchair.plah", help="指定要分析檢查的套件名稱")
    parser.add_argument("--analyze-only", help="不執行測驗，直接指定資料夾路徑進行歷史摘要重新分析")
    args = parser.parse_args()

    # 若只是要重新分析既有資料夾
    if args.analyze_only:
        target_dir = args.analyze_only
        print(f"正在重新分析資料夾：{target_dir}")
        full_path = os.path.join(target_dir, "full.txt")
        checkin_path = os.path.join(target_dir, "checkin.txt")
        deviceidle_path = os.path.join(target_dir, "deviceidle.txt")
        
        full_text = read_text(full_path)
        checkin_text = read_text(checkin_path)
        deviceidle_text = read_text(deviceidle_path)
        
        if not full_text:
            print(f"錯誤：找不到 {full_path}")
            sys.exit(1)
            
        analyze_and_print_report(full_text, checkin_text, deviceidle_text, [], [], [args.pkg])
        return

    # --- 執行完整自動化測試流程 ---
    download_dir = os.path.expanduser("~/storage/downloads")
    if not os.path.isdir(download_dir):
        print("錯誤：找不到 Termux 共享下載目錄 (~/storage/downloads)。請先執行 'termux-setup-storage'")
        sys.exit(1)

    # 1. 確保 ADB 連線
    adb_addr = ensure_adb_connected(args.port)

    # 2. 建立本次測試的輸出資料夾
    timestamp = dt.datetime.now().strftime("%Y%m%d_%H%M%S")
    out_dir = os.path.join(download_dir, "doze-test", timestamp)
    os.makedirs(out_dir, mode=0o700, exist_ok=True)
    print(f"[Doze] 測試資料將儲存至：{out_dir}")

    # 3. 測試前快照擷取
    print("[Doze] 正在擷取測試前系統狀態 (mem, battery, alarm)...")
    run_cmd(f"adb -s {adb_addr} shell dumpsys alarm > {out_dir}/alarm_start_full.txt")
    run_cmd(f"adb -s {adb_addr} shell dumpsys battery > {out_dir}/battery_start.txt")
    run_cmd(f"adb -s {adb_addr} shell dumpsys deviceidle unforce")

    # 4. 是否重置 BatteryStats
    if args.reset_batterystats:
        print("[Doze] 正在備份重置前的 BatteryStats...")
        run_cmd(f"adb -s {adb_addr} shell dumpsys batterystats > {out_dir}/batterystats_before_reset_full.txt")
        run_cmd(f"adb -s {adb_addr} shell dumpsys batterystats --reset")
        print("[Doze] BatteryStats 已重置。")

    # 5. 啟動 full-wake-history 並關閉螢幕
    run_cmd(f"adb -s {adb_addr} shell dumpsys batterystats --enable full-wake-history")
    print("[Doze] 正在發送 KEYCODE_SLEEP 關閉螢幕...")
    run_cmd(f"adb -s {adb_addr} shell input keyevent KEYCODE_SLEEP")

    print("\n" + "="*50)
    print("【Doze 測試進行中】")
    print("手機已進入關屏待機狀態。請保持靜置。")
    print("完成測試後，請喚醒手機螢幕，然後回到 Termux 按下 Enter 結束測試。")
    print("="*50 + "\n")

    try:
        input("按 Enter 鍵結束測試並收集數據 → ")
    except EOFError:
        pass

    # 6. 重新連線與收集測試後資料
    print("[Doze] 正在重新連線並擷取測試後數據...")
    ensure_adb_connected(args.port)

    run_cmd(f"adb -s {adb_addr} shell dumpsys deviceidle > {out_dir}/deviceidle.txt")
    run_cmd(f"adb -s {adb_addr} shell dumpsys battery > {out_dir}/battery_end.txt")
    run_cmd(f"adb -s {adb_addr} shell dumpsys batterystats > {out_dir}/full.txt")
    run_cmd(f"adb -s {adb_addr} shell dumpsys batterystats --checkin > {out_dir}/checkin.txt")
    run_cmd(f"adb -s {adb_addr} shell dumpsys batterystats --history > {out_dir}/history.txt")

    # 7. 呼叫內部分析引擎產生摘要
    print("[Doze] 正在分析數據並產生報告...")
    full_text = read_text(os.path.join(out_dir, "full.txt"))
    checkin_text = read_text(os.path.join(out_dir, "checkin.txt"))
    deviceidle_text = read_text(os.path.join(out_dir, "deviceidle.txt"))

    summary_path = os.path.join(out_dir, f"summary-{timestamp}.txt")
    
    # 擷取並寫入摘要
    import sys
    original_stdout = sys.stdout
    with open(summary_path, "w", encoding="utf-8") as f:
        sys.stdout = f
        analyze_and_print_report(full_text, checkin_text, deviceidle_text, [], [], [args.pkg])
        sys.stdout = original_stdout

    # 同時印在畫面上
    print(read_text(summary_path))
    print(f"\n[完成] 所有測試原始資料與摘要報告已儲存至：\n{out_dir}")


if __name__ == "__main__":
    main()
```

### 如何在 Termux 使用這支整合版工具？

1. **放置檔案**：將上面這段完整程式碼儲存為 `~/doze/doze_all-in-one.py`。
2. **授權權限**：
```bash
chmod 700 ~/doze/doze_all-in-one.py

```


3. **建立捷徑指令（方便隨時執行）**：
```bash
ln -sf ~/doze/doze_all-in-one.py "$PREFIX/bin/doze"

```


4. **日常使用方式**：
* **直接執行（自動偵測 Port）**：
```bash
doze

```


* **手動指定 Port**：
```bash
doze 39815

```


* **需要重置電池統計時**：
```bash
doze --reset-batterystats

```


* **事後重新分析既有資料夾**：
```bash
python3 ~/doze/doze_all-in-one.py --analyze-only ~/storage/downloads/doze-test/20261118_235703/

```
